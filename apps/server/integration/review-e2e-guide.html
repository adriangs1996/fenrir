Fenrir Review E2E Implementation Guide

This guide walks you through finishing apps/server/integration/review.e2e.scaffold.test.ts manually. The target is a real server integration test: real Fenrir server, real auth bootstrap, real websocket RPC, real git repository state. The only thing you control directly is the temp repo and the project/thread that point at it.

What You Already Have

  • A scoped temp repo fixture.
  • Committed, staged, unstaged, and ignored file states.
  • A real server boot helper stub.
  • Skipped it.live.skip(...) tests using Effect style.

What You Need Next

  1. Prove the server boots and answers HTTP.
  2. Bootstrap a real session with /api/auth/bootstrap.
  3. Create a typed websocket review RPC helper.
  4. Create a real project and thread bound to the temp repo.
  5. Call review.getOrCreateSession and assert the diff.

Step 3

Implement Session Bootstrap

Reuse the same HTTP bootstrap flow used by server.test.ts. Since your server config uses desktopBootstrapToken: "test-desktop-bootstrap-token", that is the credential to send.

Add a URL helper

function appendSessionCookieToWsUrl(url: string, sessionCookieHeader: string): string {
  const next = new URL(url);
  const cookie = sessionCookieHeader.split(";", 1)[0] ?? sessionCookieHeader;
  next.hash = `cookie=${encodeURIComponent(cookie)}`;
  return next.toString();
}

Replace bootstrapAuthenticatedSession

function bootstrapAuthenticatedSession(server: ReviewE2eServerHandle) {
  return Effect.gen(function* () {
    const response = yield* Effect.promise(() =>
      fetch(`${server.httpBaseUrl}/api/auth/bootstrap`, {
        method: "POST",
        headers: {
          "content-type": "application/json",
        },
        body: JSON.stringify({
          credential: "test-desktop-bootstrap-token",
        }),
      }),
    );

    assert.equal(response.ok, true);

    const sessionCookieHeader = response.headers.get("set-cookie");
    if (!sessionCookieHeader) {
      return yield* Effect.fail(new Error("Expected bootstrap to return a session cookie."));
    }

    return {
      sessionCookieHeader,
      wsUrl: appendSessionCookieToWsUrl(server.wsBaseUrl, sessionCookieHeader),
    } satisfies ReviewE2eAuthSession;
  });
}

Immediately test it

const session = yield* bootstrapAuthenticatedSession(server);
assert.equal(session.wsUrl.startsWith("ws://127.0.0.1:"), true);
Step 4

Create the Typed WebSocket RPC Helper

Copy the transport pattern from server.test.ts. You need four small helpers.

1. Parse the hash cookie back out

const parseSessionCookieFromWsUrl = (
  wsUrl: string,
): { readonly cookie: string | null; readonly url: string } => {
  const next = new URL(wsUrl);
  const cookie = next.hash.startsWith("#cookie=")
    ? decodeURIComponent(next.hash.slice("#cookie=".length))
    : null;
  next.hash = "";
  return {
    cookie,
    url: next.toString(),
  };
};

2. Build the websocket protocol layer

const wsRpcProtocolLayer = (wsUrl: string) => {
  const { cookie, url } = parseSessionCookieFromWsUrl(wsUrl);
  const webSocketConstructorLayer = Layer.succeed(
    Socket.WebSocketConstructor,
    (socketUrl, protocols) =>
      new NodeSocket.NodeWS.WebSocket(
        socketUrl,
        protocols,
        cookie ? { headers: { cookie } } : undefined,
      ) as unknown as globalThis.WebSocket,
  );

  return RpcClient.layerProtocolSocket().pipe(
    Layer.provide(Socket.layerWebSocket(url).pipe(Layer.provide(webSocketConstructorLayer))),
    Layer.provide(RpcSerialization.layerJson),
  );
};

3. Create a typed client helper

const makeWsRpcClient = RpcClient.make(WsRpcGroup);
type WsRpcClient =
  typeof makeWsRpcClient extends Effect.Effect<infer Client, any, any> ? Client : never;

const withWsRpcClient = <A, E, R>(
  wsUrl: string,
  f: (client: WsRpcClient) => Effect.Effect<A, E, R>,
) => makeWsRpcClient.pipe(
  Effect.flatMap(f),
  Effect.provide(wsRpcProtocolLayer(wsUrl)),
);

4. Replace callReviewRpc

function callReviewRpc<T>(
  session: ReviewE2eAuthSession,
  method: string,
  input: unknown,
) {
  return withWsRpcClient(session.wsUrl, (client) =>
    (
      client as unknown as Record<
        string,
        ((value: unknown) => Effect.Effect<T, Error, never>) | undefined
      >
    )[method]!(input),
  );
}

Now do a smoke check:

const config = yield* withWsRpcClient(session.wsUrl, (client) =>
  client[WS_METHODS.serverGetConfig]({}),
);

assert.equal(config.environment.label.length > 0, true);
Step 5

Create a Real Project and Thread

Do not fake the review target. The thread must point at your temp repo so that the review session resolves the same folder you seeded in the git fixture.

Use the real websocket orchestration command path. That keeps the whole flow on the real running server.

Add a small command helper

function dispatchOrchestrationCommand(
  session: ReviewE2eAuthSession,
  command: unknown,
) {
  return withWsRpcClient(session.wsUrl, (client) =>
    client[ORCHESTRATION_WS_METHODS.dispatchCommand](command),
  );
}

Replace createProjectAndThreadForRepo

function createProjectAndThreadForRepo(
  _server: ReviewE2eServerHandle,
  session: ReviewE2eAuthSession,
  workspace: ReviewE2eWorkspace,
  scenario: ReviewE2eScenario,
) {
  return Effect.gen(function* () {
    const provider = "codex";
    const projectId = ProjectId.make("review-e2e-project");
    const threadId = ThreadId.make("review-e2e-thread");
    const createdAt = new Date().toISOString();
    const defaultModel = DEFAULT_MODEL_BY_PROVIDER[provider];

    yield* dispatchOrchestrationCommand(session, {
      type: "project.create",
      commandId: CommandId.make("cmd-review-e2e-project-create"),
      projectId,
      title: "Review E2E Project",
      workspaceRoot: workspace.repoDir,
      defaultModelSelection: {
        provider,
        model: defaultModel,
      },
      createdAt,
    });

    yield* dispatchOrchestrationCommand(session, {
      type: "thread.create",
      commandId: CommandId.make("cmd-review-e2e-thread-create"),
      threadId,
      projectId,
      title: "Review E2E Thread",
      modelSelection: {
        provider,
        model: defaultModel,
      },
      interactionMode: DEFAULT_PROVIDER_INTERACTION_MODE,
      runtimeMode: "approval-required",
      branch: scenario.branchName,
      worktreePath: workspace.repoDir,
      createdAt,
    });

    return {
      projectId,
      threadId,
    } satisfies ReviewE2eThreadHandle;
  });
}

The two values that matter most are workspaceRoot: workspace.repoDir and worktreePath: workspace.repoDir. If either points somewhere else, review will inspect the wrong repository.

Step 6

Wait Until the Projection Sees the Thread

Command dispatch returns quickly, but your next review call depends on the projection snapshot already containing the project and thread. Add a small poller before calling any review RPC. 

function waitForThreadProjection(
  session: ReviewE2eAuthSession,
  threadId: ThreadId,
) {
  return Effect.gen(function* () {
    const deadline = Date.now() + 10_000;

    while (true) {
      const snapshot = yield* withWsRpcClient(session.wsUrl, (client) =>
        client[ORCHESTRATION_WS_METHODS.getBootstrapSnapshot]({}),
      );

      const threadExists = snapshot.threads.some((thread) => thread.id === threadId);
      if (threadExists) {
        return snapshot;
      }

      if (Date.now() >= deadline) {
        return yield* Effect.fail(
          new Error(`Timed out waiting for projected thread ${threadId}`),
        );
      }

      yield* Effect.sleep("50 millis");
    }
  });
}

Use it right after thread creation:

const thread = yield* createProjectAndThreadForRepo(
  server,
  session,
  workspace,
  scenario,
);

yield* waitForThreadProjection(session, thread.threadId);
Step 7

Call the Review RPCs

Once project/thread creation works, your first real review call should be review.getOrCreateSession. 

const summary = yield* callReviewRpc<{
  readonly id: string;
  readonly mode: "raw" | "review";
  readonly scope: "uncommitted" | "branch" | "combined";
  readonly fileCount: number;
  readonly chunkCount: number;
  readonly degradedReasons: readonly string[];
}>(session, "sourceControl.review.getOrCreateSession", {
  threadId: thread.threadId,
  mode: "raw",
  scope: "combined",
});

assert.equal(summary.mode, "raw");
assert.equal(summary.scope, "combined");
assert.equal(summary.fileCount > 0, true);
assert.equal(summary.chunkCount > 0, true);
assert.equal(summary.degradedReasons.includes("diff-unavailable"), false);

Then hydrate the detail surfaces:

const snapshot = yield* callReviewRpc<{
  readonly files: ReadonlyArray<{ readonly normalizedPath: string }>;
  readonly github: unknown;
}>(session, "sourceControl.review.getSessionSnapshot", {
  sessionId: summary.id,
});

const diffSnapshot = yield* callReviewRpc<{
  readonly lanes: ReadonlyArray<{
    readonly kind: string;
    readonly files: ReadonlyArray<{ readonly normalizedPath: string }>;
  }>;
}>(session, "sourceControl.review.getDiffSnapshot", {
  sessionId: summary.id,
});

Step 9

Handle the GitHub-Unavailable Case Separately

Do not start with live GitHub. First prove that local review still works when GitHub is unavailable. 

assert.equal(summary.fileCount > 0, true);
assert.equal(snapshot.files.length > 0, true);
assert.equal(summary.degradedReasons.includes("github-unavailable"), true);
assert.equal(summary.degradedReasons.includes("diff-unavailable"), false);
assert.equal(snapshot.github, null);

This is a much better first milestone than trying to get a live PR + authenticated GitHub provider stable on day one.
