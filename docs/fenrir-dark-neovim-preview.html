Port Study

Fenrir default dark mode, translated into a Neovim shell

This keeps Fenrir's near-black surfaces, cool blue-violet accent, low-contrast borders, and semantic status colors, then maps them onto editor UI primitives instead of chat UI.

source: apps/web/src/index.css

The dominant read is: black-first chrome, slightly lifted panels, translucent muted fills, bright neutral text, blue-primary interaction, and diagnostics that stay saturated without making the whole theme colorful.

// Port Fenrir dark tokens into Neovim highlight roles.
type HighlightGroup = Record<string, string>;
export function createFenrirDarkNvimTheme() {
  const palette = {
    editor: "color-mix(in srgb, #0a0a0a 95%, #ffffff)",
    lineNumber: "color-mix(in srgb, #737373 90%, #ffffff)",
    cursorLine: "color-mix(in srgb, rgb(255 255 255 / 4%) 72%, #0b0b0b)",
    keyword: "oklch(0.588 0.217 264)",
    string: "#34d399",
    diagnosticError: "#f87171",
  } as const;
  const highlights: Record<string, HighlightGroup> = {
    Normal: { fg: "#f5f5f5", bg: palette.editor },
    CursorLine: { bg: palette.cursorLine },
    Keyword: { fg: palette.keyword },
    String: { fg: palette.string },
    DiagnosticError: { fg: palette.diagnosticError },
  };
  return { palette, highlights };
}
